            <input type="time" id="reset-time-input">
            <button class="btn" onclick="saveSettings()" style="background:var(--done)">Save Settings</button>
            <button class="btn btn-red" onclick="resetAllNow()" style="margin-top:20px">Reset All Now</button>

    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getDatabase, ref, push, onValue, remove, update } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-database.js";

        const firebaseConfig = {
            apiKey: "AIzaSyAsWZsD7fhUc2PVq5GVLzP3Yl6Ql3e0EqE",
            authDomain: "task-timer---srdesign.firebaseapp.com",
            databaseURL: "https://task-timer---srdesign-default-rtdb.firebaseio.com",
            projectId: "task-timer---srdesign",
            appId: "1:175833633602:web:0a725a7ede85bdc1b42edb"
        };
        const app = initializeApp(firebaseConfig);
        const db = getDatabase(app);
        let currentTasks = {};

        onValue(ref(db, 'settings'), s => {
            if(s.val()) {
                document.getElementById('auto-reset-check').checked = s.val().auto_reset;
                document.getElementById('reset-time-input').value = s.val().reset_time;
            }
        });

        window.saveSettings = () => {
            update(ref(db, 'settings'), {
                auto_reset: document.getElementById('auto-reset-check').checked,
                reset_time: document.getElementById('reset-time-input').value
            });
        };

        window.addTask = () => {
            const title = document.getElementById('t-title').value;
            const due_time = document.getElementById('t-time').value;
            if (title && due_time) { push(ref(db, 'tasks'), { title, due_time, done: false, completedAt: "" }); document.getElementById('t-title').value = ''; }
        };

        window.addEvent = () => {
            const title = document.getElementById('e-title').value;
            const date = document.getElementById('e-date').value;
            if (title && date) { push(ref(db, 'events'), { title, date }); document.getElementById('e-title').value = ''; }
        };

        window.resetAllNow = () => { if(confirm("Reset all?")) { const up = {}; Object.keys(currentTasks).forEach(id => { up[`tasks/${id}/done`] = false; up[`tasks/${id}/completedAt`] = ""; }); update(ref(db), up); }};

        onValue(ref(db, 'tasks'), (snap) => {
            currentTasks = snap.val() || {};
            const keys = Object.keys(currentTasks);
            const done = Object.values(currentTasks).filter(t => t.done).length;
            document.getElementById('stat-total').innerText = keys.length;
            document.getElementById('stat-done').innerText = done;
            document.getElementById('stat-pending').innerText = keys.length - done;
            document.getElementById('m-tasks').innerHTML = keys.map(id => `
                <div class="item-row">
                    <div><strong>${currentTasks[id].title}</strong><br><small>${currentTasks[id].due_time}</small>
                    ${currentTasks[id].done ? `<div class="timestamp">Done at ${currentTasks[id].completedAt}</div>` : ''}</div>
                    <button class="del-btn" onclick="remove(ref(db, 'tasks/${id}'))">Del</button>
                </div>
            `).join('') || '<p>No tasks.</p>';
        });

        onValue(ref(db, 'events'), (snap) => {
            const events = snap.val() || {};
            document.getElementById('m-events').innerHTML = Object.keys(events).map(id => `
                <div class="item-row">
                    <div><strong>${events[id].title}</strong><br><small>${events[id].date}</small></div>
                    <button class="del-btn" onclick="remove(ref(db, 'events/${id}'))">Del</button>
                </div>
            `).join('') || '';
        });
    </script>
