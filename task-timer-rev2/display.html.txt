    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getDatabase, ref, onValue, update, get } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-database.js";

        const firebaseConfig = {
            apiKey: "AIzaSyAsWZsD7fhUc2PVq5GVLzP3Yl6Ql3e0EqE",
            authDomain: "task-timer---srdesign.firebaseapp.com",
            databaseURL: "https://task-timer---srdesign-default-rtdb.firebaseio.com",
            projectId: "task-timer---srdesign",
            appId: "1:175833633602:web:0a725a7ede85bdc1b42edb"
        };
        const app = initializeApp(firebaseConfig);
        const db = getDatabase(app);

        let currentTasks = [];
        let selectedIndex = 0;
        let appSettings = { auto_reset: false, reset_time: "00:00", last_reset_date: "" };

        onValue(ref(db, 'settings'), s => { if(s.val()) appSettings = {...appSettings, ...s.val()}; });

        setInterval(() => {
            const now = new Date();
            const timeStr = now.toLocaleTimeString('en-GB', { hour: '2-digit', minute: '2-digit' });
            const dateStr = now.toISOString().split('T')[0];
            document.getElementById('clock').innerText = timeStr;
            document.getElementById('date').innerText = now.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' });
            if (appSettings.auto_reset && timeStr >= appSettings.reset_time && dateStr !== appSettings.last_reset_date) {
                performAutoReset(dateStr);
            }
        }, 1000);

        async function performAutoReset(today) {
            await update(ref(db, 'settings'), { last_reset_date: today });
            const snap = await get(ref(db, 'tasks'));
            if (snap.val()) {
                const up = {};
                Object.keys(snap.val()).forEach(id => { up[`tasks/${id}/done`] = false; up[`tasks/${id}/completedAt`] = ""; });
                update(ref(db), up);
            }
        }

        onValue(ref(db, 'tasks'), (snapshot) => {
            const data = snapshot.val();
            const list = document.getElementById('task-list');
            list.innerHTML = '';
            if (data) {
                // Keep tasks in strict time order only
                currentTasks = Object.entries(data).sort((a,b) => a[1].due_time.localeCompare(b[1].due_time));
                renderTasks();
            }
        });

        function renderTasks() {
            const list = document.getElementById('task-list');
            list.innerHTML = '';
            currentTasks.forEach(([id, t], index) => {
                const div = document.createElement('div');
                div.className = `task-item ${t.done ? 'done' : ''} ${index === selectedIndex ? 'selected' : ''}`;
                div.onclick = () => handleToggle(id, t.done, index);
                div.innerHTML = `<div class="check-circle">${t.done ? '✓' : ''}</div><div><span class="task-title">${t.title}</span><span class="task-time">${t.due_time}</span></div>`;
                list.appendChild(div);
            });
        }

        function handleToggle(id, currentDone, index) {
            selectedIndex = index;
            const newStatus = !currentDone;
            const ts = newStatus ? new Date().toLocaleTimeString('en-GB', {hour:'2-digit', minute:'2-digit'}) : "";
            update(ref(db, `tasks/${id}`), { done: newStatus, completedAt: ts });
        }

        // Keyboard Logic
        window.addEventListener('keydown', (e) => {
            if (currentTasks.length === 0) return;
            if (e.key === 'ArrowDown') {
                selectedIndex = (selectedIndex + 1) % currentTasks.length;
                renderTasks();
            } else if (e.key === 'ArrowUp') {
                selectedIndex = (selectedIndex - 1 + currentTasks.length) % currentTasks.length;
                renderTasks();
            } else if (e.key === 'Enter') {
                const [id, t] = currentTasks[selectedIndex];
                const ts = new Date().toLocaleTimeString('en-GB', {hour:'2-digit', minute:'2-digit'});
                update(ref(db, `tasks/${id}`), { done: true, completedAt: ts });
            } else if (e.key === 'Backspace') {
                const [id, t] = currentTasks[selectedIndex];
                update(ref(db, `tasks/${id}`), { done: false, completedAt: "" });
            }
        });

        onValue(ref(db, 'events'), (snap) => {
            const listEl = document.getElementById('event-list');
            listEl.innerHTML = '';
            const todayStr = new Date().toISOString().split('T')[0];
            if (snap.val()) {
                const todayEvents = Object.values(snap.val()).filter(e => e.date === todayStr);
                todayEvents.forEach(e => {
                    listEl.innerHTML += `<div class="event-item"><span>${e.title}</span><span style="color:var(--accent); font-weight:700;">Today</span></div>`;
                });
                if(todayEvents.length === 0) listEl.innerHTML = '<p style="color:#475569;">No events today.</p>';
            }
        });
